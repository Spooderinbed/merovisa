- [ ] **Step 3: Create the Zod schema**

Create `lib/data/schema/nepal-police-certificate.schema.ts`:

```ts
import { z } from "zod";
import { ProvenanceSchema, HttpUrl, IsoDate } from "./common";

/**
 * Runtime schema for lib/data/source/nepal-police-certificate.ts. Guards the id + kind
 * enums, non-empty label/summary, the http(s) source, ISO lastVerified, unique ids,
 * and provenance (>=1 findingRef).
 */
const NepalPoliceCertificateRecordSchema = z.object({
  id: z.enum(["opcr-application-route", "opcr-document-set", "opcr-validity"]),
  kind: z.enum(["application-route", "required-document", "validity-rule"]),
  label: z.string().min(1),
  summary: z.string().min(1),
  source: HttpUrl,
  lastVerified: IsoDate.optional(),
  provenance: ProvenanceSchema,
});

export const NepalPoliceCertificateSchema = z
  .array(NepalPoliceCertificateRecordSchema)
  .refine((rows) => new Set(rows.map((r) => r.id)).size === rows.length, {
    message: "Nepal police-certificate ids must be unique",
  });
```

```bash
node - <<'EOF'
const fs = require('fs');
const p = 'docs/research-briefs/findings/A.jsonl';
const want = new Set(['A.095','A.096','A.097','A.100','A.102']);
const raw = fs.readFileSync(p, 'utf8');
const eol = raw.includes('\r\n') ? '\r\n' : '\n';
const lines = raw.split(/\r?\n/);
let changed = 0;
const out = lines.map((line) => {
  if (!line.trim()) return line;
  const o = JSON.parse(line);
  if (want.has(o.id)) {
    if (o.value_status !== 'unset') throw new Error(o.id + ' value_status not unset: ' + o.value_status);
    if (o.status !== 'pending') throw new Error(o.id + ' status not pending: ' + o.status);
    changed++;
    return line.replace('"value_status":"unset"', '"value_status":"prose-only"');
  }
  return line;
});
if (changed !== 5) throw new Error('expected exactly 5 changes, got ' + changed);
fs.writeFileSync(p, out.join(eol), 'utf8');
console.log('value_status -> prose-only (', changed, 'lines: A.095/096/097/100/102)');
EOF
```
