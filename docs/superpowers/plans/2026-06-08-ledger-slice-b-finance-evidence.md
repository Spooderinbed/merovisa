**Goal:** Wire five Category-B findings (B.007–B.011 — DHA's four acceptable proof-of-funds paths plus the living-cost-indicative rule) into the shipped plan, checklist, and profile finance surfaces via one new sourced data module, with every shipped phrase traced to a `used` finding and machine-checked by the reconcile harness.

**Architecture:** A new pure-data module `lib/data/source/au-financial-evidence.ts` (interface in `types.ts`, Zod schema, one registry line) holds five prose records. **Tech Stack:** TypeScript (strict, `noUncheckedIndexedAccess`), Zod, Next.js, vitest 4, `@testing-library/react`, the slice-kit tools under `docs/research-briefs/_tools/`.

- [ ] **Step 3: Create `lib/data/schema/au-financial-evidence.schema.ts`**

```ts
import { z } from "zod";
import { ProvenanceSchema, HttpUrl, IsoDate } from "./common";

/**
 * Runtime schema for lib/data/source/au-financial-evidence.ts. Guards the id +
 * kind enums, non-empty label/summary, the http(s) source, ISO lastVerified,
 * unique ids, and provenance (>=1 findingRef).
 */
const AuFinancialEvidenceRecordSchema = z.object({
  id: z.enum(["deposit", "loan", "scholarship", "parent-partner-income", "living-cost-indicative"]),
  kind: z.enum(["evidence-path", "living-cost-note"]),
  label: z.string().min(1),
  summary: z.string().min(1),
  source: HttpUrl,
  lastVerified: IsoDate.optional(),
  provenance: ProvenanceSchema,
});

export const AuFinancialEvidenceSchema = z
  .array(AuFinancialEvidenceRecordSchema)
  .refine((rows) => new Set(rows.map((r) => r.id)).size === rows.length, {
    message: "AU financial-evidence ids must be unique",
  });
```

```ts
/** Join phrases as "a, b, c, or d" (Oxford "or"). */
function oxfordOr(items: string[]): string {
  if (items.length === 0) return "";
  if (items.length === 1) return items[0]!;
  const last = items[items.length - 1]!;
  return `${items.slice(0, -1).join(", ")}, or ${last}`;
}
```
